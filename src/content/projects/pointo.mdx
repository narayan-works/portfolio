Instead of starting from speech or a chatbot panel, this project starts from pointing, shared attention, and the kind of gestures humans already use to orient one another in space.

<script is:inline>{`
    (function() {
        const sequence = [
            { gap: 0, opp: 3 },
            { gap: 1, opp: 3 },
            { gap: 2, opp: 3 },
            { gap: 3, opp: 3 },
            { gap: 0, opp: 3 },
            { gap: 1, opp: 3 },
            { gap: 2, opp: 3 },
            { gap: 2, opp: 0 },
            { gap: 2, opp: 1 },
            { gap: 2, opp: 2 },
            { gap: 2, opp: 3 },
            { gap: 2, opp: 0 },
            { gap: 2, opp: 1 },
            { gap: 2, opp: 2 },
            { gap: 2, opp: 3 },
            { gap: 3, opp: 3 }
        ];
        
        let step = 0;
        
        function cycleFrames() {
            const current = sequence[step];
            
            const gapContainer = document.getElementById('gap-gif-container');
            if (gapContainer) {
                const gapFrames = gapContainer.querySelectorAll('.gif-frame');
                gapFrames.forEach(frame => {
                    const frameIdx = parseInt(frame.getAttribute('data-frame') || '0', 10);
                    if (frameIdx === current.gap) {
                        frame.classList.add('active');
                    } else {
                        frame.classList.remove('active');
                    }
                });
            }
            
            const oppContainer = document.getElementById('opp-gif-container');
            if (oppContainer) {
                const oppFrames = oppContainer.querySelectorAll('.gif-frame');
                oppFrames.forEach(frame => {
                    const frameIdx = parseInt(frame.getAttribute('data-frame') || '0', 10);
                    if (frameIdx === current.opp) {
                        frame.classList.add('active');
                    } else {
                        frame.classList.remove('active');
                    }
                });
            }

            const finding1Container = document.getElementById('finding1-gif-container');
            if (finding1Container) {
                const finding1FrameIdx = step % 2;
                const finding1Frames = finding1Container.querySelectorAll('.gif-frame');
                finding1Frames.forEach(frame => {
                    const frameIdx = parseInt(frame.getAttribute('data-frame') || '0', 10);
                    if (frameIdx === finding1FrameIdx) {
                        frame.classList.add('active');
                    } else {
                        frame.classList.remove('active');
                    }
                });
            }
            
            step = (step + 1) % sequence.length;
        }
        
        setInterval(cycleFrames, 2000);
    })();
`}</script>
